01. مقدمة في قواعد البيانات وعالم SQL

فهم بنية قواعد البيانات العلائقية (Relational Databases) وأهمية لغة SQL في إدارتها.

-- لغة هيكلة الاستعلامات القياسية
SELECT VERSION();

02. إنشاء قواعد البيانات CREATE DATABASE

تخصيص قاعدة بيانات جديدة مع ضبط ترميز الحروف لدعم اللغة العربية (utf8mb4).

CREATE DATABASE technical_knowledge 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

03. حذف وإدارة قواعد البيانات DROP

أوامر حذف وإزالة قواعد البيانات الخاملة بشكل نهائي من الخادم.

DROP DATABASE IF EXISTS old_database;

04. إنشاء الجداول CREATE TABLE

تصميم الجداول وتحديد الأعمدة وأنواع البيانات الهندسية المناسبة لها.

CREATE TABLE users (
    id INT AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE
);

05. أنواع البيانات في SQL

استعراض أنواع البيانات الرقمية والنصية والتاريخية مثل INT, VARCHAR, TEXT, DATE.

-- أمثلة لأنواع البيانات الشائعة
-- INT للأرقام، VARCHAR للنصوص القصيرة، TEXT للنصوص الطويلة

06. المفتاح الأساسي PRIMARY KEY

تحديد عمود فريد وغير مكرر لتعريف كل سجل داخل الجدول بتميز تام.

CREATE TABLE books (
    book_id INT PRIMARY KEY,
    title VARCHAR(200)
);

07. القيود constraints (NOT NULL, UNIQUE)

فرض قواعد أمان تمنع ترك الحقول فارغة أو تكرار القيم الخاصة.

CREATE TABLE authors (
    id INT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE
);

08. القيمة الافتراضية DEFAULT

إسناد قيمة تلقائية للحقل في حال لم يتم إدخالها أثناء عملية الإضافة.

CREATE TABLE settings (
    id INT PRIMARY KEY,
    status VARCHAR(20) DEFAULT 'active'
);

09. المفتاح الأجنبي FOREIGN KEY

ربط الجداول ببعضها البعض لضمان السلامة المرجعية (Referential Integrity).

CREATE TABLE articles (
    article_id INT PRIMARY KEY,
    author_id INT,
    FOREIGN KEY (author_id) REFERENCES authors(id)
);

10. تعديل الجداول ALTER TABLE

إضافة، حذف، أو تعديل الأعمدة والقيود في جدول موجود مسبقاً.

ALTER TABLE users ADD COLUMN phone VARCHAR(20);

11. حذف الجداول DROP & TRUNCATE

الفرق بين مسح الجدول بالكامل أو تفريغ محتوياته مع الاحتفاظ بالهيكل.

TRUNCATE TABLE logs;

12. إدخال البيانات INSERT INTO

إضافة سجلات جديدة وبيانات فعلية داخل أعمدة الجداول.

INSERT INTO users (name, email) 
VALUES ('مها شهوب', 'maha@tech.ly');

13. الاستعلام الأساسي SELECT

جلب واستعراض البيانات من الجداول المحددة بكل مرونة.

SELECT name, email FROM users;

14. شرط التصفية WHERE

تحديد السجلات المطلوبة فقط بناءً على شروط منطقية محددة.

SELECT * FROM users WHERE id = 1;

15. المعاملات المنطقية (AND, OR, NOT)

دمج شروط متعددة لتصفية دقيقة ومتقدمة للبيانات.

SELECT * FROM books WHERE price < 50 AND category = 'Programming';

16. ترتيب النتائج ORDER BY

تنظيم شكل عرض النتائج تصاعدياً (ASC) أو تنازلياً (DESC).

SELECT * FROM articles ORDER BY created_at DESC;

17. تحديد عدد النتائج LIMIT

اقتصار عرض النتائج على عدد معين من السجلات أو لغايات التقسيم (Pagination).

SELECT * FROM users LIMIT 10;

18. تصفية القيم الفريدة DISTINCT

إزالة التكرارات وعرض القيم الفريدة فقط من الأعمدة.

SELECT DISTINCT category FROM books;

19. البحث النمطي باستخدام LIKE

البحث عن النصوص المطابقة لأنماط معينة باستخدام الأحرف البديلة (مثل %).

SELECT * FROM books WHERE title LIKE '%أساسيات%';

20. معامل النطاق BETWEEN

جلب القيم الواقعة ضمن نطاق محدد بين حد أدنى وأقصى.

SELECT * FROM products WHERE price BETWEEN 100 AND 500;

21. معامل القوائم IN

التحقق مما إذا كانت القيمة مطابقة لأي عنصر ضمن قائمة محددة.

SELECT * FROM users WHERE city IN ('طرابلس', 'بنغازي', 'مصراتة');

22. التعامل مع القيم الفارغة IS NULL

البحث وفحص السجلات التي تحوي قيماً فارغة (NULL) في حقل معين.

SELECT * FROM users WHERE phone IS NULL;

23. تحديث البيانات UPDATE

تعديل وتحديث القيم المخزنة مسبقاً في السجلات بناءً على الشروط.

UPDATE users SET status = 'active' WHERE id = 5;

24. حذف السجلات DELETE

إزالة سجلات معينة من الجداول مع إمكانية استخدام شروط التصفية.

DELETE FROM logs WHERE created_at < '2025-01-01';

25. دوال التجميع الأساسية COUNT, SUM, AVG

حساب أعداد السجلات، مجموع القيم، والمعدلات الحسابية للأعمدة.

SELECT COUNT(*), AVG(price) FROM products;

26. الدوال العددية الصغرى والكبرى MIN, MAX

استخراج أقل وأكبر قيمة رقمية أو تاريخية من حقل معين.

SELECT MIN(price), MAX(price) FROM products;

27. تجميع البيانات GROUP BY

تجميع السجلات المتشابهة لتطبيق دوال التجميع عليها لكل مجموعة.

SELECT category, COUNT(*) FROM books GROUP BY category;

28. تصفية التجمعات HAVING

فلترة النتائج الناتجة عن عمليات التجميع GROUP BY (على عكس WHERE).

SELECT category, COUNT(*) FROM books GROUP BY category HAVING COUNT(*) > 5;

29. الربط الداخلي INNER JOIN

دمج جدولين أو أكثر استناداً إلى عمود مشترك وجلب البيانات المتاحة في الطرفين.

SELECT articles.title, users.name 
FROM articles 
INNER JOIN users ON articles.author_id = users.id;

30. الربط اليساري LEFT JOIN

جلب كافة السجلات من الجدول الأيسر حتى وإن لم توجد مطابقة في الجدول الأيمن.

SELECT users.name, articles.title 
FROM users 
LEFT JOIN articles ON users.id = articles.author_id;

31. الربط اليميني RIGHT JOIN

جلب كافة السجلات من الجدول الأيمن بغض النظر عن وجود مطابقة في الأيسر.

SELECT articles.title, categories.name 
FROM articles 
RIGHT JOIN categories ON articles.cat_id = categories.id;

32. الربط الكامل FULL OUTER JOIN

عرض كافة السجلات من الجدولين عند توفر مطابقة في أي منهما.

SELECT * FROM table1 
FULL OUTER JOIN table2 ON table1.id = table2.ref_id;

33. الربط الذاتي Self Join

ربط الجدول بنفسه لمقارنة الصفوف الداخلية (مثل علاقة المدير والموظف).

SELECT e1.name AS employee, e2.name AS manager 
FROM employees e1 
LEFT JOIN employees e2 ON e1.manager_id = e2.id;

34. الدمج باستخدام UNION

دمج نتائج استعلامين أو أكثر في جدول نتائج واحد مع إزالة التكرار.

SELECT email FROM users UNION SELECT email FROM clients;

35. الاستعلامات الفرعية Subqueries

تضمين استعلام داخل استعلام آخر لمعالجة شروط معقدة ومتداخلة.

SELECT * FROM products WHERE price > (SELECT AVG(price) FROM products);

36. المعاملات EXISTS و NOT EXISTS

التحقق من وجود سجلات مطابقة ضمن الاستعلامات الفرعية بكفاءة عالية.

SELECT * FROM users u WHERE EXISTS (SELECT * FROM orders o WHERE o.user_id = u.id);

37. الدوال النصية String Functions

معالجة النصوص عبر الحروف الكبيرة والصغيرة والدمج والقص (CONCAT, SUBSTRING).

SELECT CONCAT(first_name, ' ', last_name) AS full_name FROM employees;

38. الدوال الرياضية Math Functions

إجراء العمليات الحسابية المتقدمة كالتقريب الجذر التربيعي والقيمة المطلقة.

SELECT ROUND(price, 2) FROM products;

39. دوال التواريخ والأوقات Date Functions

استخراج السنة والشهر واليوم وحساب الفترات الزمنية بين التواريخ.

SELECT NOW(), DATEDIFF(expiration_date, NOW()) FROM subscriptions;

40. المعامل الشرطي CASE WHEN

تطبيق الشروط البرمجية المنطقية داخل استعلامات SQL وتوليد أعمدة ديناميكية.

SELECT name, 
    CASE WHEN score >= 90 THEN 'ممتاز' ELSE 'جيد' END AS rating 
FROM students;

41. تحويل الأنواع CAST و CONVERT

تحويل أنواع بيانات الحقول مؤقتاً أثناء عرض واستعلام النتائج.

SELECT CAST('2026-06-01' AS DATE);

42. معالجة القيم النل COALESCE

إرجاع أول قيمة غير فارغة من قائمة القيم المُمررة.

SELECT COALESCE(phone, mobile, 'غير متوفر') FROM users;

43. إنشاء الفهارس Indexes لتسريع البحث

بناء فهارس على الأعمدة التي يتم البحث بها بكثرة لمضاعفة سرعة الأداء.

CREATE INDEX idx_user_email ON users(email);

44. الحذف الآلي للفهارس DROP INDEX

إزالة الفهارس غير المستخدمة لتخفيف العبء على عمليات التعديل والإدخال.

DROP INDEX idx_user_email ON users;

45. الجداول الوهمية Views

حفظ استعلام معقد كجدول افتراضي يتم استدعاؤه بسهولة تامة.

CREATE VIEW active_users AS SELECT * FROM users WHERE status = 'active';

46. إدارة الصلاحيات GRANT و REVOKE

منح وسحب صلاحيات الوصول وقراءة وتعديل البيانات للمستخدمين.

GRANT SELECT, INSERT ON db.* TO 'maha'@'localhost';

47. المعاملات المالية Transactions (COMMIT & ROLLBACK)

ضمان تنفيذ مجموعة أوامر معاً بنجاح أو التراجع عنها بالكامل عند حدوث خطأ.

START TRANSACTION;
UPDATE accounts SET balance = balance - 100 WHERE id = 1;
COMMIT;

48. الإجراءات المخزنة Stored Procedures

تخزين مجموعات الأوامر المعقدة داخل الخادم لتنفيذها بنداء مباشر.

CREATE PROCEDURE GetActiveUsers()
BEGIN
    SELECT * FROM users WHERE status = 'active';
END;

49. الدوال المعرفة من المستخدم Functions

تصميم دوال مخصصة تعيد قيمة أو نتيجة حسابية محددة للاستعلامات.

CREATE FUNCTION CalcTax(amount DECIMAL) RETURNS DECIMAL
RETURN amount * 0.15;

50. المشغلات Triggers

تنفيذ أوامر تلقائية فور حدوث عمليات إدخال أو تحديث أو حذف على الجداول.

CREATE TRIGGER before_insert_user 
BEFORE INSERT ON users 
FOR EACH ROW SET NEW.status = 'active';

EXPLAIN SELECT * FROM users WHERE email = 'test@test.com';

إنشاء حسابات المستخدمين وتأمين كلمات المرور الخاصة بالوصول للقاعدة.

CREATE USER 'admin_user'@'localhost' IDENTIFIED BY 'SecurePass2026';

58. التعامل مع البيانات الكبيرة JSON في SQL

تخزين واستعلام بيانات بصيغة JSON داخل قواعد بيانات SQL العلائقية الحديثة.

SELECT settings->>'$.theme' FROM users;

59. النوافذ التحليلية Window Functions (ROW_NUMBER)

ترقيم الصفوف وحساب العمليات ضمن نوافذ بيانات محددة دون تجميع الجدول.

SELECT name, salary, ROW_NUMBER() OVER(ORDER BY salary DESC) as row_num FROM employees;

60. الدوال الترتيبية RANK و DENSE_RANK

توزيع الرتب والدرجات على السجلات مع التعامل مع القيم المتطابقة.

SELECT score, RANK() OVER(ORDER BY score DESC) FROM exams;

61. مقارنة القيم السابقة واللاحقة LAG و LEAD

جلب قيم الصفوف السابقة أو اللاحقة للمقارنة التحليلية المباشرة.

SELECT salary, LAG(salary, 1) OVER(ORDER BY id) FROM employees;

62. الجداول المؤقتة Temporary Tables

إنشاء جداول مؤقتة لتخزين العمليات الحسابية الوسطية وتدمج تلقائياً بانتهاء الجلسة.

CREATE TEMPORARY TABLE temp_summary AS SELECT * FROM sales WHERE year = 2026;

63. الجداول الشائعة المشتركة CTEs

استخدام تعبيرات الجداول المشتركة (WITH) لكتابة استعلامات أكثر نظافة وهندسة.

WITH TopSales AS (SELECT * FROM sales WHERE amount > 1000)
SELECT * FROM TopSales;

64. الاستعلامات التكرارية Recursive CTEs

معالجة البيانات الهرمية والشجرية عبر الاستعلامات المتكررة ذاتياً.

WITH RECURSIVE nums AS (SELECT 1 as n UNION ALL SELECT n+1 FROM nums WHERE n < 5)
SELECT * FROM nums;

استخدام جدول وسيط (Junction Table) لربط كيانين بعلاقة متعدد لمتعدد.

CREATE TABLE student_courses (student_id INT, course_id INT);

67. إنشاء جداول الأرشيف التاريخية

فصل السجلات القديمة وأرشفتها لتخفيف أحمال قاعدة البيانات الحية.

CREATE TABLE orders_archive LIKE orders;

68. تحسين استعلامات البحث النصي الكامل Full-Text Search
