MySQL هو نظام إدارة قواعد بيانات علائقية RDBMS يستخدم لغة SQL لتخزين البيانات وتنظيمها والاستعلام عنها وإدارتها. يستخدم في المواقع والتطبيقات وأنظمة المؤسسات والـAPIs وتطبيقات سطح المكتب.

MySQL Server

03 — تثبيت MySQL

يمكن تشغيل MySQL عبر MySQL Server أو MySQL Installer أو XAMPP أو Docker، بحسب بيئة العمل. المنفذ الافتراضي الشائع هو 3306 ويمكن تغييره.

06 — إنشاء وإدارة قواعد البيانات

CREATE DATABASE IF NOT EXISTS school;

USE school;

تحذير: DROP DATABASE يحذف قاعدة البيانات ومحتوياتها، لذلك يجب التأكد من وجود نسخة احتياطية قبل التنفيذ في نظام حقيقي.

07 — الجداول

CREATE TABLE students (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(150),
    age INT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

الجدول هو الوحدة الأساسية لتخزين السجلات داخل قاعدة البيانات.

11 — Foreign Keys

CREATE TABLE departments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL
);

CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    department_id INT,
    FOREIGN KEY (department_id)
        REFERENCES departments(id)
);

23 — JOIN

SELECT
    e.name AS employee_name,
    d.name AS department_name
FROM employees e
INNER JOIN departments d
    ON e.department_id = d.id;
JOINالفكرة
INNER JOINالسجلات المطابقة فقط.
LEFT JOINكل سجلات الجدول الأيسر مع المطابق.
RIGHT JOINكل سجلات الجدول الأيمن مع المطابق.
CROSS JOINكل التركيبات الممكنة بين الصفوف.

24 — GROUP BY

SELECT department_id, COUNT(*) AS employee_count
FROM employees
GROUP BY department_id;

25 — HAVING

SELECT department_id, COUNT(*) AS employee_count
FROM employees
GROUP BY department_id
HAVING COUNT(*) > 5;

WHERE تصفي الصفوف قبل التجميع، بينما HAVING تصفي نتائج التجميع.

27 — Subqueries

SELECT *
FROM students

    FROM students

الاستعلامات الفرعية مفيدة، لكن أحيانًا يمكن إعادة صياغتها باستخدام JOIN أو CTE للحصول على وضوح أو أداء أفضل بحسب الحالة.

33 — Triggers

الـTrigger ينفذ تلقائيًا عند حدث مثل INSERT أو UPDATE أو DELETE.

CREATE TRIGGER before_student_insert
BEFORE INSERT ON students
FOR EACH ROW
SET NEW.name = TRIM(NEW.name);
استخدم Triggers باعتدال لأنها تضيف سلوكًا تلقائيًا قد يصعب تتبعه في المشاريع الكبيرة.

35 — ACID

Atomicity
المعاملة كوحدة واحدة.
Consistency
الحفاظ على سلامة القواعد.

تنظيم تداخل المعاملات.

36 — Isolation Levels

المستوىالفكرة
READ UNCOMMITTEDأقل عزلًا.
READ COMMITTEDقراءة البيانات الملتزم بها.
REPEATABLE READيحافظ على اتساق القراءة ضمن المعاملة وفق سلوك المحرك.
SERIALIZABLEأعلى عزلًا عادةً مع تكلفة تزامن أعلى.

يجب فهم تأثير المستوى على Dirty Reads وNon-repeatable Reads وPhantom Reads والتزامن.

37 — Locks وDeadlocks

توجد آليات قفل مختلفة مثل Shared Locks وExclusive Locks وRow Locks وMetadata Locks. يحدث Deadlock عندما تنتظر معاملات بعضها بعضًا بطريقة تمنع التقدم.

صمم المعاملات قصيرة قدر الإمكان، ووحّد ترتيب الوصول إلى الموارد، وتعامل مع احتمالية إعادة المحاولة في التطبيقات التي تحتاج ذلك.

38 — Indexes

CREATE INDEX idx_students_name
ON students(name);

الفهرس قد يسرع البحث والفرز والربط، لكنه يستهلك مساحة ويضيف تكلفة لبعض عمليات الكتابة.

39 — Composite Indexes

CREATE INDEX idx_name_age
ON students(name, age);

ترتيب الأعمدة داخل الفهرس المركب مهم، ويجب تصميمه وفق الاستعلامات الفعلية.

41 — Query Optimization

  • اختيار الأعمدة المطلوبة بدل SELECT * عندما يكون ذلك مناسبًا.
  • إنشاء فهارس مبنية على الاستعلامات الفعلية.
  • تحليل EXPLAIN وEXPLAIN ANALYZE.
  • تقليل حجم البيانات المنقولة.
  • تحسين JOIN وWHERE.
  • استخدام Pagination.
  • مراقبة الاستعلامات البطيئة.

42 — Normalization

التطبيع يقلل تكرار البيانات ومشاكل الإدخال والتحديث والحذف.

المستوىالفكرة العامة
1NFقيم ذرية وبنية مناسبة للحقول.
2NFإزالة الاعتماد الجزئي في الجداول ذات المفاتيح المركبة.
3NFإزالة الاعتماد الانتقالي غير الضروري.
BCNFصيغة أكثر صرامة من التطبيع.

قد تستخدم Denormalization بصورة مدروسة لأسباب أداء أو تقارير.

44 — Character Sets

Character Set يحدد كيفية تمثيل النصوص. في التطبيقات الحديثة متعددة اللغات يعد utf8mb4 خيارًا شائعًا ومناسبًا.

CREATE DATABASE school
CHARACTER SET utf8mb4;

45 — Collations

Collation تحدد قواعد مقارنة وترتيب النصوص. اختيارها يؤثر في المقارنة والترتيب وحساسية حالة الأحرف وبعض القواعد اللغوية.

CREATE DATABASE school
CHARACTER SET utf8mb4
COLLATE utf8mb4_0900_ai_ci;
تحقق من Collation المتاحة في إصدار MySQL الذي تستخدمه، لأن الخيارات تختلف حسب الإصدار.

46 — اللغة العربية وutf8mb4

للتعامل مع العربية تأكد من توافق الترميز بين قاعدة البيانات والجدول والاتصال والتطبيق. لا يكفي تغيير ترميز الجدول فقط إذا كان الاتصال نفسه مضبوطًا بطريقة غير صحيحة.

CREATE TABLE articles (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(255) NOT NULL,
    content TEXT
) CHARACTER SET utf8mb4;

47 — JSON

CREATE TABLE products (
    id INT PRIMARY KEY,
    data JSON

INSERT INTO products (id, data)
VALUES (1, '{"name":"Laptop","ram":16}');

JSON مناسب لبعض البيانات المرنة، لكن لا ينبغي استخدامه تلقائيًا بدل التصميم العلائقي عندما تكون البيانات الأساسية منظمة.

48 — CTE

WITH adult_students AS (
    SELECT *
    FROM students

49 — Recursive CTE

Recursive CTE مفيد للبيانات الهرمية مثل الموظفين والمديرين أو الأشجار.

WITH RECURSIVE numbers AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1
    FROM numbers
    WHERE n < 5
)
SELECT * FROM numbers;

50 — Window Functions

SELECT
    name,
    age,
    ROW_NUMBER() OVER (ORDER BY age DESC) AS row_num,
    RANK() OVER (ORDER BY age DESC) AS ranking
FROM students;

ومن الدوال المهمة: ROW_NUMBER وRANK وDENSE_RANK وLAG وLEAD ودوال التجميع مع OVER.

51 — FULLTEXT

CREATE FULLTEXT INDEX ft_articles
ON articles(title, content);

SELECT *
FROM articles
WHERE MATCH(title, content)
AGAINST('database');

يستخدم للبحث النصي المتخصص، مع مراعاة إعدادات اللغة ومحرك التخزين وطبيعة المحتوى.

SELECT * FROM temp_students;

الجداول المؤقتة مرتبطة عادةً بجلسة الاتصال وتفيد في العمليات الوسيطة.

53 — Generated Columns

يمكن إنشاء عمود مشتق من تعبير أو أعمدة أخرى. يفيد في بعض حالات التصفية والفهرسة وتبسيط الوصول إلى قيم مشتقة.

CREATE TABLE products (
    price DECIMAL(10,2),
    quantity INT,
    total DECIMAL(12,2)
      GENERATED ALWAYS AS (price * quantity) STORED
);

57 — SQL Injection

يحدث SQL Injection عندما يتم دمج مدخل المستخدم داخل SQL بطريقة غير آمنة.

-- لا تفعل:
SELECT * FROM users
WHERE name = '...مدخل المستخدم...';

-- استخدم Prepared Statement في التطبيق:
SELECT * FROM users
WHERE name = ?;

58 — Prepared Statements

تفصل القيم عن نص SQL وتقلل خطر SQL Injection وتفيد في إعادة استخدام الاستعلام.

SELECT *
FROM users
WHERE email = ?;
في VB.NET أو PHP أو Python استخدم API الخاصة بالمكتبة لإضافة Parameters بدل بناء النص يدويًا.

59 — Backup

النسخ الاحتياطي قد يكون منطقيًا أو فيزيائيًا حسب الأداة والبنية. من الأدوات المعروفة mysqldump وMySQL Shell، وتوجد حلول أخرى للمؤسسات.

mysqldump -u root -p school > school.sql

60 — Restore

الاستعادة جزء أساسي من خطة النسخ الاحتياطي. يجب اختبارها دوريًا بدل افتراض أن ملف النسخة الاحتياطية صالح.

mysql -u root -p school < school.sql

61 — Binary Log

Binary Log يسجل تغييرات البيانات والعمليات المناسبة وفق إعدادات الخادم، ويدخل في Replication وبعض سيناريوهات الاستعادة الزمنية.

إدارة الـBinary Logs تحتاج سياسة احتفاظ ومساحة تخزين مناسبة.

62 — Replication

Replication ينسخ البيانات بين خوادم MySQL. من المفاهيم المهمة Source وReplica وBinary Log وGTID.

63 — High Availability

من حلول MySQL الحديثة للتوافر العالي: Group Replication وInnoDB Cluster وMySQL Router وMySQL Shell، بحسب البنية المطلوبة.

التوافر العالي ليس مجرد نسخ البيانات؛ يجب تصميم Failover والمراقبة والنسخ الاحتياطي واختبارات التعافي.

64 — Monitoring

راقب CPU وRAM والتخزين والاتصالات والاستعلامات والأقفال والـSlow Queries والـReplication والـBuffer Pool والأخطاء.

65 — Slow Query Log

يساعد Slow Query Log في اكتشاف الاستعلامات البطيئة وتحليل أسبابها مثل الفهارس المفقودة أو الاستعلامات غير الفعالة.

حدد عتبة مناسبة للاستعلام البطيء وفق طبيعة النظام، ثم حلل النتائج بدل زيادة العتبة لإخفاء المشكلة.

66 — Performance Schema

Performance Schema يوفر بيانات داخلية لمراقبة وتحليل أداء MySQL، مثل بعض الاستعلامات والأحداث والانتظار والموارد.

SELECT *
FROM performance_schema.setup_instruments
LIMIT 10;

67 — sys Schema

sys Schema يوفر Views مساعدة مبنية على بيانات المراقبة لتسهيل تحليل الأداء والاستخدام.

SELECT *
FROM sys.schema_table_statistics
LIMIT 10;

68 — MySQL Shell

69 — Docker

70 — MySQL مع PHP

$pdo = new PDO(
    'mysql:host=localhost;dbname=school;charset=utf8mb4',
    $user,
    $password

    'SELECT * FROM students WHERE id = ?'

$stmt->execute([$id]);

71 — MySQL مع VB.NET

يمكن استخدام MySqlConnector أو MySql.Data. المثال التالي يستخدم MySqlConnector.

Imports MySqlConnector

Using con As New MySqlConnection(connectionString)
    Await con.OpenAsync()

    Using cmd As New MySqlCommand(
        "SELECT id, name FROM students WHERE age >= @age", con)

        cmd.Parameters.AddWithValue("@age", 18)

        Using reader = Await cmd.ExecuteReaderAsync()
            While Await reader.ReadAsync()
                Console.WriteLine(reader("name").ToString())
            End While
        End Using
    End Using
End Using

78 — Connection Pooling

إنشاء اتصال جديد لكل طلب قد يكون مكلفًا. مكتبات الاتصال الحديثة توفر Connection Pooling لإعادة استخدام الاتصالات وفق إعدادات التطبيق.

اضبط حجم الـPool والمهلات حسب عدد الطلبات والموارد وسلوك التطبيق، ولا تترك الاتصالات مفتوحة بلا حاجة.
